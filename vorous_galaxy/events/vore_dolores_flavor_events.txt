# voragons.170 — The Intern Problem
# Security has filed its third incident report this month. She has been actively feeding.
# Hunger resets in immediate — the eating has already been happening.
country_event = {
	id = voragons.170
	title = voragons.170.name
	desc = voragons.170.desc
	picture = GFX_evt_DOLORES_vore
	show_sound = event_vore_gulp
	is_triggered_only = yes

	trigger = {
		exists = event_target:Gov_dolores
	}

	immediate = {
		# She's been actively feeding all quarter — hunger resets regardless of player response
		dolores_hunger_reset_effect = yes
	}

	# No action: morale erodes quietly, but the administrative simplicity saves something
	option = {
		name = voragons.170.a
		custom_tooltip = voragons.170.a.tooltip
		add_resource = { unity = -50 }
		add_resource = { energy = 100 }
	}

	# Implement safety protocols: costs unity + energy; restricting her access
	# means hunger builds faster going forward
	option = {
		name = voragons.170.b
		custom_tooltip = voragons.170.b.tooltip
		add_resource = { unity = -100 }
		add_resource = { energy = -150 }
		hidden_effect = {
			set_country_flag = dolores_safety_protocols
			dolores_hunger_increase_effect = yes
		}
	}
}

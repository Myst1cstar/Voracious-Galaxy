###################################
#   DOLORES RANDOM LEADER EVENTS
#
#   Triggered by on_monthly_pulse_country (~6%/month, 180-day cooldown).
#   These events represent Dolores acting on her own initiative —
#   sometimes with warning, sometimes without.
#
#   voragons.200 — Priority Target    (warning: she has filed a schedule)
#   voragons.201 — Post-Breakfast     (aftermath: a scientist is gone)
#   voragons.202 — Helpful Suggestion (offer: let her handle a commander)
#   voragons.203 — Tuesday            (flavor: unnamed staff, no warning)
#   voragons.204 — Record Time        (humor: someone ran fast enough)
###################################

# voragons.200 — Priority Target
# Dolores has formally designated one of your leaders as her next meal.
# Player gets advance warning and a choice.
country_event = {
	id = voragons.200
	title = voragons.200.name
	desc = voragons.200.desc
	picture = GFX_evt_DOLORES_vore
	show_sound = dolores_paragon_sound_effects
	is_triggered_only = yes

	trigger = {
		exists = event_target:Gov_dolores
		NOT = { has_country_flag = dolores_random_event_cooldown }
		any_owned_leader = {
			NOT = { is_same_value = event_target:Gov_dolores }
		}
	}

	immediate = {
		set_timed_country_flag = { flag = dolores_random_event_cooldown days = 365 }
		# Save a random non-Dolores leader as her target
		random_owned_leader = {
			limit = {
				NOT = { is_same_value = event_target:Gov_dolores }
			}
			save_event_target_as = dolores_target
		}
	}

	# Warn the target and post a guard — leader survives, but Dolores grows hungrier
	option = {
		name = voragons.200.a
		custom_tooltip = voragons.200.a.tooltip
		add_resource = { influence = -75 }
		hidden_effect = { dolores_hunger_increase_effect = yes }
	}

	# Let the schedule proceed — Dolores is pleased, hunger resets
	option = {
		name = voragons.200.b
		custom_tooltip = voragons.200.b.tooltip
		hidden_effect = {
			event_target:dolores_target = {
				kill_leader = { show_notification = yes }
			}
			dolores_hunger_reset_effect = yes
		}
		add_resource = { society_research = 800 }
	}

	# Tell her to find her own targets without the paperwork — frustration builds
	option = {
		name = voragons.200.c
		custom_tooltip = voragons.200.c.tooltip
		add_resource = { unity = -150 }
		hidden_effect = { dolores_hunger_increase_effect = yes }
	}
}

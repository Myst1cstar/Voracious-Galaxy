# voragons.155 — She ate the ruler.
# Fires 5000 days after Dolores is hired (set in voragons.150 Recruit option).
# Player chooses what to do about the now-headless empire.
country_event = {
	id = voragons.155
	title = voragons.155.name
	desc = voragons.155.desc
	picture = GFX_evt_DOLORES_vore
	show_sound = event_vore_gulp
	is_triggered_only = yes

	immediate = {
		# Specifically target the empire's current ruler — not just any commander.
		# The old `kill_leader = { class = commander }` could pick any commander-class
		# leader at random, which often left the actual ruler alive.
		random_owned_leader = {
			limit = {
				is_ruler = yes
				NOT = { has_leader_flag = Dolores }
			}
			save_event_target_as = previous_ruler
			kill_leader = { show_notification = yes }
		}
		dolores_hunger_reset_effect = yes
		# A ruler is a substantial meal — empire gets the Just Fed bonus
		dolores_just_fed_effect = yes
	}

	option = {
		name = voragons.155.a
		custom_tooltip = voragons.155.a.tooltip
		hidden_effect = {
			country_event = { id = voragons.157 }
		}
	}

	option = {
		name = voragons.155.b
		custom_tooltip = voragons.155.b.tooltip
		hidden_effect = {
			country_event = { id = voragons.156 days = 25 }
		}
	}

	option = {
		name = voragons.155.c
		custom_tooltip = voragons.155.c.tooltip
		hidden_effect = {
			# Step 1: Constitutional reform (the previous ruler was eaten above).
			# Skip Hive Minds and Machine Intelligences — forcing them to a personal
			# authority would break their core mechanics.
			if = {
				limit = {
					NOR = {
						has_authority = auth_hive_mind
						has_authority = auth_machine_intelligence
						has_authority = auth_ancient_machine_intelligence
						has_authority = auth_dictatorial
					}
				}
				change_government = {
					authority = auth_dictatorial
					civics = {
						civic = random
						civic = random
					}
					cooldown = no
				}
			}

			add_resource = { unity = 500 }

			# Step 2 is DELAYED to voragons.218 (fires in 5 days). change_government
			# does not propagate within the same effect tick — the new dictatorial
			# council positions (councilor_ruler_dictatorial) do not exist yet, so
			# set_council_position would fail with "position doesn't exist in government".
			# Waiting a few days lets the reform fully apply before we seat Dolores.
			country_event = { id = voragons.218 days = 5 }

			# The cosmogenesis path opens regardless — she has plans either way.
			country_event = { id = voragons.157 }
		}
	}
}

# voragons.218 — Seats Dolores as ruler AFTER the government reform has propagated.
# Split out from voragons.155.c because change_government needs a tick to apply
# before the authority-specific ruler council position exists.
country_event = {
	id = voragons.218
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		# Pick the ruler council position matching the now-current authority.
		# set_council_position is the valid 4.x effect (leader scope).
		if = {
			limit = { has_authority = auth_hive_mind }
			event_target:Gov_dolores = { set_council_position = councilor_ruler_hive_mind }
		}
		else_if = {
			limit = {
				OR = {
					has_authority = auth_machine_intelligence
					has_authority = auth_ancient_machine_intelligence
				}
			}
			event_target:Gov_dolores = { set_council_position = councilor_ruler_machine_intelligence }
		}
		else_if = {
			limit = { has_authority = auth_imperial }
			event_target:Gov_dolores = { set_council_position = councilor_ruler_imperial }
		}
		else_if = {
			limit = { has_authority = auth_democratic }
			event_target:Gov_dolores = { set_council_position = councilor_ruler_democratic }
		}
		else_if = {
			limit = { has_authority = auth_oligarchic }
			event_target:Gov_dolores = { set_council_position = councilor_ruler_oligarchic }
		}
		else_if = {
			limit = { has_authority = auth_corporate }
			event_target:Gov_dolores = { set_council_position = councilor_ruler_corporate }
		}
		else = {
			# Default: dictatorial (the result of the reform in voragons.155.c)
			event_target:Gov_dolores = { set_council_position = councilor_ruler_dictatorial }
		}

		# Flag + hub: enables ruler arc events and succession crisis alternative
		set_country_flag = dolores_is_ruler
		country_event = { id = voragons.219 }
	}
}

# Gorger
country_event = {
	id = vore_leader_event.10
	is_triggered_only = yes

	trigger = {
			exists = ruler
			from.species = {
				OR = {
			is_archetype = VOREBIOLOGICAL
			is_archetype = VOREMACHINE
				}
		}
	}

	immediate = {
		random_owned_leader = {
			save_event_target_as = gorger_to_be
		}
		}

	picture = "GFX_vore_gorger"
	title = vore_leader_event.10.name
	desc = vore_leader_event.10.desc

	option = {
		name = vore_leader_event.10.acknowledged

		event_target:gorger_to_be = {
			add_trait = leader_trait_vore_gorger
		}
	}
}

##############################################
#			VORE TRAIT LEVEL UP
##############################################

# LEADER FLAG vore_leveled_up PREVENTS EVENTS ACTIVATING IN CHAIN FOR THE SAME LEADER AT ONCE
# THEY DON'T WORK IF LEADER LEVELS UP MORE THAN ONCE WITHIN 2 DAYS

##########
# ADMIRAL
##########

# Novice to apprentice
country_event{
	id = vore_leader_event.20
	is_triggered_only = yes
	
	trigger = {
		from = {
			has_trait = trait_admiral_novice_predator
			NOT = {
				has_leader_flag = vore_leveled_up
			}
		}
	}

	title = vore_leader_event.20.name
	desc = vore_leader_event.20.desc

	immediate = {
		from = {
			remove_trait = trait_admiral_novice_predator
			add_trait = { trait = trait_admiral_apprentice_predator	}
			set_leader_flag = vore_leveled_up
			leader_event = {
				id = vore_leader_event.50
				days = 2
			}
		}
	}

	option = {
		name = DELICIOUS
	}
}

# REMOVES THE vore_leveled_up LEADER FLAG
# USED SO ALL LEVEL UP EVENTS DONT TRIGGER IN SUCCESSION
leader_event{
	id = vore_leader_event.50
	is_triggered_only = yes
	hide_window = yes

	immediate = {
		this = {
			remove_leader_flag = vore_leveled_up
		}
	}
}
